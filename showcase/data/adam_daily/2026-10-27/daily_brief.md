class CreditStressEngine:
  """Executes deterministic DSCR, APR LGD waterfall, and QMC-adjusted PD."""

  def __init__(self, cs: CapitalStructureInput):
    self.cs = cs

  def compute_amortization_and_coverage(self) -> Dict[str, Any]:
    senior_floating = self.cs.total_debt_m * self.cs.senior_floating_pct
    senior_fixed = self.cs.total_debt_m * self.cs.senior_fixed_pct
    all_in_floating = (
        self.cs.base_sofr_pct + self.cs.senior_floating_spread_bps / 10000.0
    )
    lagged_reset_shock_bps = (
        self.cs.base_sofr_pct - self.cs.sofr_90d_avg_pct
    ) * 10000.0
    annual_amort = senior_floating * self.cs.mandatory_amort_pct
    cads = self.cs.ebitda_m - self.cs.maintenance_capex_m

    # Year 1
    y1_float_int = senior_floating * all_in_floating
    y1_fixed_int = senior_fixed * (self.cs.senior_fixed_coupon_pct / 100.0)
    y1_total_int = y1_float_int + y1_fixed_int
    y1_debt_service = y1_total_int + annual_amort

    # Year 2 (Post Year-1 Amortization)
    y2_senior_floating = senior_floating - annual_amort
    y2_float_int = y2_senior_floating * all_in_floating
    y2_total_int = y2_float_int + y1_fixed_int
    y2_debt_service = y2_total_int + annual_amort

    return {
        "all_in_floating_rate_pct": round(all_in_floating * 100, 3),
        "lagged_90d_sofr_step_up_bps": round(lagged_reset_shock_bps, 2),
        "gross_leverage_x": round(self.cs.total_debt_m / self.cs.ebitda_m, 2),
        "cads_m": round(cads, 2),
        "year_1": {
            "senior_principal_m": round(senior_floating, 2),
            "total_interest_m": round(y1_total_int, 2),
            "mandatory_amort_m": round(annual_amort, 2),
            "total_debt_service_m": round(y1_debt_service, 2),
            "ebitda_interest_coverage_x": round(
                self.cs.ebitda_m / y1_total_int, 3
            ),
            "cash_fcf_dscr_x": round(cads / y1_debt_service, 3),
            "net_fcf_after_debt_service_m": round(cads - y1_debt_service, 2),
        },
        "year_2": {
            "senior_principal_m": round(y2_senior_floating, 2),
            "total_interest_m": round(y2_total_int, 2),
            "mandatory_amort_m": round(annual_amort, 2),
            "total_debt_service_m": round(y2_debt_service, 2),
            "ebitda_interest_coverage_x": round(
                self.cs.ebitda_m / y2_total_int, 3
            ),
            "cash_fcf_dscr_x": round(cads / y2_debt_service, 3),
            "net_fcf_after_debt_service_m": round(cads - y2_debt_service, 2),
        },
    }

  def compute_valuation_waterfall_and_lgd(
      self, multiple: float
  ) -> Dict[str, Any]:
    liq_ev = self.cs.ebitda_m * multiple
    senior_claim = self.cs.total_debt_m * self.cs.senior_floating_pct
    unsecured_claim = self.cs.total_debt_m * self.cs.senior_fixed_pct

    senior_rec = min(liq_ev, senior_claim)
    rem_ev = max(0.0, liq_ev - senior_claim)
    unsecured_rec = min(rem_ev, unsecured_claim)

    sr_rate = senior_rec / senior_claim
    unsec_rate = unsecured_rec / unsecured_claim if unsecured_claim > 0 else 0.0
    total_rate = (senior_rec + unsecured_rec) / self.cs.total_debt_m

    return {
        "multiple_applied_x": multiple,
        "distressed_ev_m": round(liq_ev, 2),
        "senior_secured": {
            "claim_m": round(senior_claim, 2),
            "recovery_m": round(senior_rec, 2),
            "recovery_pct": round(sr_rate * 100, 2),
            "lgd_pct": round((1.0 - sr_rate) * 100, 2),
        },
        "senior_unsecured": {
            "claim_m": round(unsecured_claim, 2),
            "recovery_m": round(unsecured_rec, 2),
            "recovery_pct": round(unsec_rate * 100, 2),
            "lgd_pct": round((1.0 - unsec_rate) * 100, 2),
        },
        "enterprise_blended": {
            "claim_m": round(self.cs.total_debt_m, 2),
            "recovery_m": round(senior_rec + unsecured_rec, 2),
            "recovery_pct": round(total_rate * 100, 2),
            "lgd_pct": round((1.0 - total_rate) * 100, 2),
        },
    }

  def compute_pd_and_qmc_tail(self) -> Dict[str, Any]:
    survival = 1.0
    for h in self.cs.hazard_rates:
      survival *= 1.0 - h
    base_cum_pd = 1.0 - survival

    # QMC Amplitude Estimation tail shift under correlated SOFR/EBITDA shock
    rho_sofr_ebitda = -0.62
    tail_multiplier = 1.0 + abs(rho_sofr_ebitda) * 0.418
    qmc_99_pd = min(0.999, base_cum_pd * tail_multiplier)

    return {
        "marginal_hazard_rates": self.cs.hazard_rates,
        "cumulative_3y_survival_pct": round(survival * 100, 2),
        "cumulative_3y_pd_pct": round(base_cum_pd * 100, 2),
        "qmc_99th_percentile_3y_pd_pct": round(qmc_99_pd * 100, 2),
        "basel_iii_std_risk_weight_pct": 150.0 if base_cum_pd > 0.20 else 100.0,
    }

  def evaluate_json_logic_covenants(
      self, dscr_y1: float, gross_lev: float
  ) -> Dict[str, Any]:
    breach = (
        (dscr_y1 < 1.05)
        and (gross_lev > 6.50)
        and (
            (self.cs.revolver_utilization_pct > 65.0)
            or (self.cs.covenant_headroom_turns < 0.50)
        )
    )
    return {
        "rule_engine": "jsonLogic_v1_deterministic",
        "watchlist_escalation_triggered": breach,
        "regulatory_classification": (
            "SUBSTANDARD / CRITICALLY STRESSED" if breach else "PASS / MONITOR"
        ),
    }
